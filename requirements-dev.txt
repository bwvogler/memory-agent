pytest>=8.0.0

# Linter and type checker, pinned exactly rather than floored.
#
# A new ruff release can add rules to `ALL`, and ty is pre-1.0 (0.0.x) with
# diagnostics that still move between releases. Either one drifting turns a
# pull request red for something nobody in that pull request changed, which
# teaches everyone to ignore CI. Bump these deliberately, and update
# .github/workflows/ci.yml in the same commit. There is no
# .pre-commit-config.yaml in this repo - scripts/pre-commit-checks.sh reads
# these pins directly instead of duplicating them.
ruff==0.15.18
ty==0.0.71

# Headless browser driver for the `browser-test` skill
# (.claude/skills/browser-test) - verifying frontend/JS behavior that no
# Python-only test tier can see, since none of them execute the served
# JavaScript. Pinned for the same reason as ruff/ty: each release bundles a
# specific Chromium build, so an unpinned bump could silently change what
# actually runs. After bumping, re-run `playwright install chromium`.
playwright==1.62.0
